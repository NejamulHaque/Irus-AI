POST /api/v1/chat

Body: {"message": "...", "web_search": false} → returns {"reply": "..."}. Limit: 30 req/min.

curl -X POST {{ url_for('main.api_chat', _external=True) }} \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"message": "Hello Irus!", "web_search": false}'
import requests

r = requests.post(
    "{{ url_for('main.api_chat', _external=True) }}",
    headers={"Authorization": "Bearer YOUR_API_KEY"},
    json={"message": "Hello Irus!", "web_search": False},
)
print(r.json()["reply"])
const r = await fetch("{{ url_for('main.api_chat', _external=True) }}", {
  method: "POST",
  headers: { "Authorization": "Bearer YOUR_API_KEY",
             "Content-Type": "application/json" },
  body: JSON.stringify({ message: "Hello Irus!" })
});
console.log((await r.json()).reply);
